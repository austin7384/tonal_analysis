% 9. Devil's advocate
% ---------------------------------------------------------------------------
\begin{frame}{Identification concern: GPT-5 may introduce systematic measurement bias}
    \vfill
    \begin{columns}[T, onlytextwidth]

        \begin{column}{0.44\textwidth}
            \RaggedRight
            \colorbox{LightGray}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
                \vspace{0.12cm}\hspace{0.2cm}
                {\normalsize\bfseries\color{WarmGray} Potential confound}
                \vspace{0.08cm}
            }}
            \vspace{0.35cm}
            {\normalsize
            GPT-5 may assign systematically\\
            higher scores to abstracts by\\
            female-majority teams, independent\\
            of underlying linguistic content.}
        \end{column}

        \begin{column}{0.08\textwidth}\end{column}

        \begin{column}{0.48\textwidth}
            \RaggedRight
            \colorbox{LightGray}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
                \vspace{0.12cm}\hspace{0.2cm}
                {\normalsize\bfseries\color{Teal} Response}
                \vspace{0.08cm}
            }}
            \vspace{0.35cm}
            {\normalsize
            \textbf{(1)} LLM readability replicates Hengel's\\
            \hspace{0.5cm}manually calculated scores.\\[0.3cm]
            \textbf{(2)} Results are consistent across six\\
            \hspace{0.5cm}authorship specifications.\\[0.3cm]
            \textbf{(3)} A model exhibiting such bias would\\
            \hspace{0.5cm}produce elevated scores on assertiveness\\
            \hspace{0.5cm}and hedging. The data show no such pattern.
            }
        \end{column}

    \end{columns}
    \vfill
\end{frame}
