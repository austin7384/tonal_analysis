\begin{frame}[shrink=10]{Scientific writing is not a neutral container for ideas}
    \vfill
    \begin{columns}[T, onlytextwidth]

            \colorbox{LightGray}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{%

                \vspace{0.06cm}
            }}
            \vspace{0.2cm}
            {\normalsize
            \begin{itemize}\setlength\itemsep{0.08cm}
                \item How research is framed shapes peer evaluation and knowledge dissemination
                \item Gilligan (1993): dominant standards may overlook different modes of communication
                \item Economics literature focuses on publication and promotion outcomes, not the writing itself
            \end{itemize}
            }
        \end{column}

        \begin{column}{0.04\textwidth}\end{column}

        \begin{column}{0.48\textwidth}
            \RaggedRight
            \colorbox{LightGray}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{%

                {\small\bfseries\color{Teal} This paper}
                \vspace{0.06cm}
            }}
            \vspace{0.2cm}
            {\normalsize
            \begin{itemize}\setlength\itemsep{0.08cm}
                \item Does author-team gender composition predict systematic differences in how economists write?
                \item Builds on Hengel (2022): female-authored papers are more readable; writing gap widens during peer review
                \item Extends beyond readability to \textbf{15 additional} writing dimensions
                \item 9,117 abstracts $\cdot$ 16-item LLM rubric
            \end{itemize}
            }
        \end{column}

    \end{columns}
    \vfill
\end{frame}

% 6. LLM Evaluations
% ---------------------------------------------------------------------------
\begin{frame}{GPT-5 with chain-of-thought reasoning scored all 9,117 abstracts}
    \vfill
    {\normalsize
    \begin{itemize}\setlength\itemsep{0.3cm}
        \item \textbf{Model:} GPT-5 — extended reasoning before answering improves accuracy on complex tasks (Shojaee et al.\ 2025)
        \item \textbf{Pipeline:} abstract $\rightarrow$ rubric $\rightarrow$ model $\rightarrow$ structured 16-score output
        \item \textbf{Prompt engineering:} step-by-step reasoning induced per criterion; shorter inputs increase accuracy (Modarressi et al.\ 2025)
        \item \textbf{Validation:} trained on subset, tested on holdout, then validated against Hengel (2022) readability scores
        \item {\color{WarmGray}\textit{Caveat:} some randomness in LLM outputs; reported SEs may understate true uncertainty}
    \end{itemize}
    }
    \vfill
\end{frame}

\begin{frame}{Three-step analysis: validate $\rightarrow$ mean differences $\rightarrow$ controlled regressions}

        \textbf{(1)} Benchmark LLM readability against Hengel (2022) measures\quad
        \textbf{(2)} Unconditional mean differences across all 16 criteria\quad
        \textbf{(3)} Controlled regressions
        }
    \end{center}
    \vspace{0.4cm}
    \begin{center}
        {\large
        $R_j = \beta_0 + \beta_1 \cdot \textit{FemaleRatio}_j
             + \gamma X_j + \delta Y_j + \varepsilon_j$}

    \vspace{0.4cm}
    \begin{columns}[T, onlytextwidth]
